43SaturnBurger::SaturnBurger(Mesh mesh, BurgersOptions options)
44 : mMesh(std::move(mesh)),
45 mOptions(std::move(options)),
46 mPart(mMesh.nx, mMesh.ny, mMesh.nz, mOptions.periodic && mMesh.nx > 1,
47 mOptions.periodic && mMesh.dimension >= 2 && mMesh.ny > 1,
48 mOptions.periodic && mMesh.dimension >= 3 && mMesh.nz > 1),
49 mLocal(LocalMesh::distributed(mMesh, mPart, 2)),
50 mHalo(mPart, 2, 1),
51 mFieldHalo(mHalo, 1),
52 mU(mPart, 2, 1, "saturnBurger::u"),
53 mFlux(mPart, 2, 1, "saturnBurger::flux"),
54 mLapFace(mPart, 2, 1, "saturnBurger::lap_face"),
55 mDiv(mPart, 2, 1, "saturnBurger::div"),
56 mLap(mPart, 2, 1, "saturnBurger::lap"),
57 mGrad(mPart, 2, 3, "saturnBurger::grad") {
58 if (!mMesh.structured || mMesh.dimension < 1 || mMesh.dimension > 3)
59 throw std::invalid_argument(
60 "saturnBurger: expected a structured 1D, 2D or 3D mesh");
61 if (mMesh.dimension == 1 && (mMesh.ny != 1 || mMesh.nz != 1))
62 throw std::invalid_argument("saturnBurger: 1D requires ny = nz = 1");
63 if (mMesh.dimension == 2 && mMesh.nz != 1)
64 throw std::invalid_argument("saturnBurger: 2D requires nz = 1");
65 if (mMesh.nx < 1 || mMesh.ny < 1 || mMesh.nz < 1)
66 throw std::invalid_argument("saturnBurger: empty mesh");
67 if (mOptions.nu < 0)
68 throw std::invalid_argument("saturnBurger: negative diffusivity");
69 if (!mOptions.initial)
70 throw std::invalid_argument("saturnBurger: initial condition is missing");
71 if (!mOptions.periodic && !mOptions.dirichlet)
72 throw std::invalid_argument(
73 "saturnBurger: non-periodic boundaries need a Dirichlet callback");
74 Real minH = std::numeric_limits<Real>::max();
75 const Real exponent = Real(1) / static_cast<Real>(mMesh.dimension);
76 for (int k = mPart.localOrigin(2);
77 k < mPart.localOrigin(2) + mPart.localExtent(2); ++k) {
78 for (int j = mPart.localOrigin(1);
79 j < mPart.localOrigin(1) + mPart.localExtent(1); ++j) {
80 for (int i = mPart.localOrigin(0);
81 i < mPart.localOrigin(0) + mPart.localExtent(0); ++i)
82 minH = std::min(minH, std::pow(mLocal.cellVolume(i, j, k), exponent));
83 }
84 }
85 if (!(minH > 0))
86 throw std::invalid_argument("saturnBurger: non-positive cell size");
87 mMinCellSize = minH;
88}

94void SaturnBurger::applyInitialConditions() {
95 auto host = mU.hostView();
96 for (int k = mPart.localOrigin(2);
97 k < mPart.localOrigin(2) + mPart.localExtent(2); ++k) {
98 for (int j = mPart.localOrigin(1);
99 j < mPart.localOrigin(1) + mPart.localExtent(1); ++j) {
100 for (int i = mPart.localOrigin(0);
101 i < mPart.localOrigin(0) + mPart.localExtent(0); ++i) {
102 const int ii = i - (mU.origin(0) - mU.nGrow());
103 const int jj = j - (mU.origin(1) - mU.nGrow());
104 const int kk = k - (mU.origin(2) - mU.nGrow());
105 host(ii, jj, kk, 0) = mOptions.initial(mLocal.cellCentroid(i, j, k));
106 }

110 applyBoundaryConditions();

113void SaturnBurger::applyBoundaryConditions() {
114 mFieldHalo.execute(mU);
115 if (mOptions.periodic)
116 return;
117

119 const Real time = mTime;

121 conditions[p].type = BCType::Dirichlet;
122 conditions[p].value = [this, time](const Vec3& x, const Vec3&, int c) {
123 return mOptions.dirichlet(x, time, c);
124 };
125 }
126 fillBoundaryGhosts(mU, mLocal, conditions);

129void SaturnBurger::rhs(const CellField& state, CellField& rate) {
130 if (state.ncomp() != 1)
131 throw std::invalid_argument("saturnBurger: expected a scalar state");
132 burgersFlux(state, mFlux, mLocal, mOptions.convection, mOptions.limiter);
133 divergence(mFlux, mDiv, mLocal);
134 laplacian(state, mLapFace, mGrad, mLap, mLocal);
135
136 CellField r = rate;
137 CellField div = mDiv;
138 CellField lap = mLap;
139 const Real nu = mOptions.nu;
140 const Real source = mOptions.source;
141 Kokkos::parallel_for(
142 "saturnBurger::rhs", ownedPolicy(state),
143 KOKKOS_LAMBDA(int i, int j, int k) {
144 r(i, j, k, 0) = -div(i, j, k, 0) + nu * lap(i, j, k, 0) + source;
145 });
146}

148std::vector<ProfilePoint> SaturnBurger::profile(Real time,
149 bool withExact) const {
150 if (withExact && !mOptions.exact)
151 throw std::runtime_error(
152 "saturnBurger: profile comparison needs an exact solution");
153
154 const int nx = mU.extent(0);
155 const int ny = mU.extent(1);
156 const int nz = mU.extent(2);
157 const int i0 = mU.origin(0);
158 const int j0 = mU.origin(1);
159 const int k0 = mU.origin(2);
160 const int n = nx * ny * nz;
161
162 // Pack x,y,z,u_numerical,u_exact,cell_id per cell into one flat buffer so
163 // the global profile needs a single MPI_Allgatherv.
164 std::vector<double> local(static_cast<std::size_t>(6) * n);
165 auto host = mU.hostView();
166 const int gi0 = mU.origin(0) - mU.nGrow();
167 const int gj0 = mU.origin(1) - mU.nGrow();
168 const int gk0 = mU.origin(2) - mU.nGrow();
169 int a = 0;
170 for (int k = k0; k < k0 + nz; ++k) {
171 for (int j = j0; j < j0 + ny; ++j) {
172 for (int i = i0; i < i0 + nx; ++i) {
173 const int cell = mLocal.cellId(i, j, k);
174 if (cell < 0)
175 throw std::runtime_error("saturnBurger: missing owned cell");
176 const Vec3 centre = mLocal.cellCentroid(i, j, k);
177 const std::size_t base = static_cast<std::size_t>(6) * a;
178 local[base + 0] = centre.x;
179 local[base + 1] = centre.y;
180 local[base + 2] = centre.z;
181 local[base + 3] = host(i - gi0, j - gj0, k - gk0, 0);
182 local[base + 4] = withExact ? mOptions.exact(centre, time) : 0.0;
183 local[base + 5] = static_cast<double>(cell);
184 ++a;
185 }
186 }
187 }
188
189 std::vector<double> global;
190#ifdef SATURN_USE_MPI
191 int localCount = n;
192 int worldSize = 1;
193 MPI_Comm_size(MPI_COMM_WORLD, &worldSize);
194 std::vector<int> counts(static_cast<std::size_t>(worldSize));
195 std::vector<int> offsets(static_cast<std::size_t>(worldSize));
196 MPI_Allgather(&localCount, 1, MPI_INT, counts.data(), 1, MPI_INT,
197 MPI_COMM_WORLD);
198 offsets[0] = 0;
199 for (int r = 1; r < worldSize; ++r)
200 offsets[static_cast<std::size_t>(r)] =
201 offsets[static_cast<std::size_t>(r - 1)] +
202 counts[static_cast<std::size_t>(r - 1)];
203 const int total = offsets.back() + counts.back();
204 const int localDoubles = 6 * n;
205 std::vector<int> doubleCounts(static_cast<std::size_t>(worldSize));
206 std::vector<int> doubleOffsets(static_cast<std::size_t>(worldSize));
207 for (int r = 0; r < worldSize; ++r) {
208 doubleCounts[static_cast<std::size_t>(r)] =
209 6 * counts[static_cast<std::size_t>(r)];
210 doubleOffsets[static_cast<std::size_t>(r)] =
211 6 * offsets[static_cast<std::size_t>(r)];
212 }
213 global.resize(static_cast<std::size_t>(6) * total);
214 MPI_Allgatherv(local.data(), localDoubles, MPI_DOUBLE, global.data(),
215 doubleCounts.data(), doubleOffsets.data(), MPI_DOUBLE,
216 MPI_COMM_WORLD);
217#else
218 global = std::move(local);
219#endif
220
221 const std::size_t pointCount = global.size() / 6;
222 std::vector<ProfilePoint> points(pointCount);
223 for (std::size_t p = 0; p < pointCount; ++p) {
224 const std::size_t base = 6 * p;
225 points[p].cell = static_cast<Index>(global[base + 5]);
226 points[p].x = global[base + 0];
227 points[p].y = global[base + 1];
228 points[p].z = global[base + 2];
229 points[p].numerical = global[base + 3];
230 points[p].exact = global[base + 4];
231 points[p].error = points[p].numerical - points[p].exact;
232 }
233 std::sort(points.begin(), points.end(),
234 [](const ProfilePoint& a, const ProfilePoint& b) {
235 if (a.x != b.x)
236 return a.x < b.x;
237 if (a.y != b.y)
238 return a.y < b.y;
239 return a.z < b.z;
240 });
241 return points;
242}

244ExactError SaturnBurger::exactError(Real time) const {
245 const std::vector<ProfilePoint> points = profile(time);
246 ExactError out;
247 out.samples = static_cast<BigIndex>(points.size());
248 if (points.empty())
249 return out;
250 double l1 = 0;
251 double l2 = 0;
252 double linf = 0;
253 for (const ProfilePoint& p : points) {
254 const double e = static_cast<double>(p.error);
255 l1 += std::abs(e);
256 l2 += e * e;
257 linf = std::max(linf, std::abs(e));
258 }
259 out.l1 = static_cast<Real>(l1 / static_cast<double>(points.size()));
260 out.l2 =
261 static_cast<Real>(std::sqrt(l2 / static_cast<double>(points.size())));
262 out.linf = static_cast<Real>(linf);
263 return out;
264}

303Real SaturnBurger::maxWaveSpeed() const {
304 CellField u = mU;
