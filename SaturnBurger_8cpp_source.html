43SaturnBurger::SaturnBurger(Mesh mesh, BurgersOptions options)
44 : mMesh(std::move(mesh)),
45 mOptions(std::move(options)),
46 mPart(mMesh.nx, mMesh.ny, mMesh.nz, mOptions.periodic, false, false),
47 mLocal(mMesh, mPart, 2),
48 mHalo(mPart, 2, 1),
49 mFieldHalo(mHalo, 1),
50 mU(mPart, 2, 1, "saturnBurger::u"),
51 mFlux(mPart, 2, 1, "saturnBurger::flux"),
52 mLapFace(mPart, 2, 1, "saturnBurger::lap_face"),
53 mDiv(mPart, 2, 1, "saturnBurger::div"),
54 mLap(mPart, 2, 1, "saturnBurger::lap"),
55 mGrad(mPart, 2, 3, "saturnBurger::grad") {
56 if (!mMesh.structured || mMesh.dimension != 1 || mMesh.ny != 1 ||
57 mMesh.nz != 1)
58 throw std::invalid_argument(
59 "saturnBurger: expected a structured 1D mesh (ny = nz = 1)");
60 if (mMesh.nx < 1)
61 throw std::invalid_argument("saturnBurger: empty mesh");
62 if (mOptions.nu < 0)
63 throw std::invalid_argument("saturnBurger: negative diffusivity");
64 if (!mOptions.initial)
65 throw std::invalid_argument("saturnBurger: initial condition is missing");
66 if (!mOptions.periodic && !mOptions.dirichlet)
67 throw std::invalid_argument(
68 "saturnBurger: non-periodic boundaries need a Dirichlet callback");
69 if (mMesh.cellVolume.size() != mMesh.cells.size() ||
70 mMesh.cellCentroid.size() != mMesh.cells.size())
71 throw std::invalid_argument(
72 "saturnBurger: computeGeometry() must run before the solver");
73
74 Real minH = std::numeric_limits<Real>::max();
75 for (Real volume : mMesh.cellVolume)
76 minH = std::min(minH, volume);
77 if (!(minH > 0))
78 throw std::invalid_argument("saturnBurger: non-positive cell size");
79 mMinCellSize = minH;
80}

86void SaturnBurger::applyInitialConditions() {
87 auto host = mU.hostView();
88 for (int k = mPart.localOrigin(2);
89 k < mPart.localOrigin(2) + mPart.localExtent(2); ++k) {
90 for (int j = mPart.localOrigin(1);
91 j < mPart.localOrigin(1) + mPart.localExtent(1); ++j) {
92 for (int i = mPart.localOrigin(0);
93 i < mPart.localOrigin(0) + mPart.localExtent(0); ++i) {
94 const int cell = mMesh.cellAt(i, j, k);
95 if (cell < 0)
96 throw std::runtime_error("saturnBurger: missing owned cell");
97 const int ii = i - (mU.origin(0) - mU.nGrow());
98 const int jj = j - (mU.origin(1) - mU.nGrow());
99 const int kk = k - (mU.origin(2) - mU.nGrow());
100 host(ii, jj, kk, 0) = mOptions.initial(mMesh.cellCentroid[cell]);
101 }
102 }
103 }
104 mU.syncFromHost(host);
105 applyBoundaryConditions();
106}

108void SaturnBurger::applyBoundaryConditions() {
109 mFieldHalo.execute(mU);
110 if (mOptions.periodic)
111 return;
112
113 std::vector<BoundaryCondition> conditions(mMesh.patches.size());
114 const Real time = mTime;
115 for (std::size_t p = 0; p < conditions.size(); ++p) {
116 conditions[p].type = BCType::Dirichlet;
117 conditions[p].value = [this, time](const Vec3& x, const Vec3&, int c) {
118 return mOptions.dirichlet(x, time, c);
119 };
120 }
121 fillBoundaryGhosts(mU, mLocal, conditions);

124void SaturnBurger::rhs(const CellField& state, CellField& rate) {
125 if (state.ncomp() != 1)
126 throw std::invalid_argument("saturnBurger: expected a scalar state");
127 burgersFlux(state, mFlux, mLocal, mOptions.convection, mOptions.limiter);
128 divergence(mFlux, mDiv, mLocal);
129 laplacian(state, mLapFace, mGrad, mLap, mLocal);
130
131 CellField r = rate;
132 CellField div = mDiv;
133 CellField lap = mLap;
134 const Real nu = mOptions.nu;
135 const Real source = mOptions.source;
136 Kokkos::parallel_for(
137 "saturnBurger::rhs", ownedPolicy(state),
138 KOKKOS_LAMBDA(int i, int j, int k) {
139 r(i, j, k, 0) = -div(i, j, k, 0) + nu * lap(i, j, k, 0) + source;
140 });
141}

143std::vector<ProfilePoint> SaturnBurger::profile(Real time) const {
144 if (!mOptions.exact)
145 throw std::runtime_error(
146 "saturnBurger: profile comparison needs an exact solution");
147
148 const int i0 = mPart.localOrigin(0);
149 const int j0 = mPart.localOrigin(1);
150 const int k0 = mPart.localOrigin(2);
151 const int n = mPart.localExtent(0);
152
153 std::vector<double> x(static_cast<std::size_t>(n));
154 std::vector<double> numerical(static_cast<std::size_t>(n));
155 std::vector<double> exact(static_cast<std::size_t>(n));
156
157 auto host = mU.hostView();
158 const int gi0 = mU.origin(0) - mU.nGrow();
159 const int gj0 = mU.origin(1) - mU.nGrow();
160 const int gk0 = mU.origin(2) - mU.nGrow();
161 for (int a = 0; a < n; ++a) {
162 const int i = i0 + a;
163 const int cell = mMesh.cellAt(i, j0, k0);
164 if (cell < 0)
165 throw std::runtime_error("saturnBurger: missing owned cell");
166 const Vec3& centre = mMesh.cellCentroid[cell];
167 x[static_cast<std::size_t>(a)] = centre.x;
168 numerical[static_cast<std::size_t>(a)] =
169 host(i - gi0, j0 - gj0, k0 - gk0, 0);
170 exact[static_cast<std::size_t>(a)] = mOptions.exact(centre, time);
171 }
172
173 std::vector<double> globalX;
174 std::vector<double> globalNumerical;
175 std::vector<double> globalExact;
176#ifdef SATURN_USE_MPI
177 int localCount = n;
178 int worldSize = 1;
179 MPI_Comm_size(MPI_COMM_WORLD, &worldSize);
180 std::vector<int> counts(static_cast<std::size_t>(worldSize));
181 std::vector<int> offsets(static_cast<std::size_t>(worldSize));
182 MPI_Allgather(&localCount, 1, MPI_INT, counts.data(), 1, MPI_INT,
183 MPI_COMM_WORLD);
184 offsets[0] = 0;
185 for (int r = 1; r < worldSize; ++r)
186 offsets[static_cast<std::size_t>(r)] =
187 offsets[static_cast<std::size_t>(r - 1)] +
188 counts[static_cast<std::size_t>(r - 1)];
189 const int total = offsets.back() + counts.back();
190 globalX.resize(static_cast<std::size_t>(total));
191 globalNumerical.resize(static_cast<std::size_t>(total));
192 globalExact.resize(static_cast<std::size_t>(total));
193 MPI_Allgatherv(x.data(), localCount, MPI_DOUBLE, globalX.data(),
194 counts.data(), offsets.data(), MPI_DOUBLE, MPI_COMM_WORLD);
195 MPI_Allgatherv(numerical.data(), localCount, MPI_DOUBLE,
196 globalNumerical.data(), counts.data(), offsets.data(),
197 MPI_DOUBLE, MPI_COMM_WORLD);
198 MPI_Allgatherv(exact.data(), localCount, MPI_DOUBLE, globalExact.data(),
199 counts.data(), offsets.data(), MPI_DOUBLE, MPI_COMM_WORLD);
200#else
201 globalX = std::move(x);
202 globalNumerical = std::move(numerical);
203 globalExact = std::move(exact);
204#endif
205
206 std::vector<ProfilePoint> points(globalX.size());
207 for (std::size_t a = 0; a < globalX.size(); ++a) {
208 points[a].x = globalX[a];
209 points[a].numerical = globalNumerical[a];
210 points[a].exact = globalExact[a];
211 points[a].error = points[a].numerical - points[a].exact;
212 }
213 std::sort(
214 points.begin(), points.end(),
215 [](const ProfilePoint& a, const ProfilePoint& b) { return a.x < b.x; });
216 return points;
217}

219ExactError SaturnBurger::exactError(Real time) const {
220 const std::vector<ProfilePoint> points = profile(time);
221 ExactError out;
222 out.samples = static_cast<BigIndex>(points.size());
223 if (points.empty())
224 return out;
225 double l1 = 0;
226 double l2 = 0;
227 double linf = 0;
228 for (const ProfilePoint& p : points) {
229 const double e = static_cast<double>(p.error);
230 l1 += std::abs(e);
231 l2 += e * e;
232 linf = std::max(linf, std::abs(e));
233 }
234 out.l1 = static_cast<Real>(l1 / static_cast<double>(points.size()));
235 out.l2 =
236 static_cast<Real>(std::sqrt(l2 / static_cast<double>(points.size())));
237 out.linf = static_cast<Real>(linf);
238 return out;
239}
