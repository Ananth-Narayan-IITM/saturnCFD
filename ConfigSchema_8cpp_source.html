124void ConfigSchema::validate(const Config& cfg) const {
125 // Units first: non-SI is rejected before any value is interpreted.
126 (void)cfg.unitSystem();
127
128 for (const auto& key : cfg.leafKeys()) {
129 if (key == "units.system") {
130 if (mOptions.find(key) == mOptions.end())
131 throw ConfigError("units.system is not declared by this schema");
132 continue;
133 }
134 if (mOptions.find(key) == mOptions.end())
135 throw ConfigError("unknown key '" + key + "' in '" + cfg.source() + "'");
136 }
137
138 for (const auto& [key, opt] : mOptions) {
139 if (key == "units.system")
140 continue;
141 if (!cfg.has(key)) {
142 if (opt.required)
143 throw ConfigError("missing required key '" + key + "' in '" +
144 cfg.source() + "'");
145 continue;
146 }
147
148 switch (opt.type) {
149 case OptionType::String:
150 (void)cfg.getString(key, "");
151 break;
152 case OptionType::Bool:
153 (void)cfg.getBool(key, false);
154 break;
155 case OptionType::Int: {
156 const int v = cfg.getInt(key, 0);
157 if (opt.hasRange && (v < opt.minValue || v > opt.maxValue))
158 throw ConfigError("key '" + key + "' is out of range");
159 break;
160 }
161 case OptionType::Double: {
162 const Real v = cfg.getDouble(key, 0);
163 if (opt.hasRange && (v < opt.minValue || v > opt.maxValue))
164 throw ConfigError("key '" + key + "' is out of range");
165 break;
166 }
167 case OptionType::Enum: {
168 const std::string v = cfg.getString(key, "");
169 if (!inEnum(v, opt.enumValues)) {
170 std::string allowed;
171 for (std::size_t i = 0; i < opt.enumValues.size(); ++i) {
172 if (i)
173 allowed += " | ";
174 allowed += opt.enumValues[i];
175 }
176 throw ConfigError("key '" + key + "' has invalid value '" + v +
177 "' (expected " + allowed + ")");
178 }
179 break;
180 }
181 }
182 }
183}

185void ConfigSchema::dump(std::ostream& os, const std::string& solverName) const {
186 os << "solver: " << solverName << "\n\n";
187
188 std::string previousGroup;
189 for (const auto& [key, opt] : mOptions) {
190 if (key == "solver")
191 continue;
192 const auto pos = key.rfind('.');
193 const std::string group =
194 pos == std::string::npos ? "" : key.substr(0, pos);
195 const std::string leaf =
196 pos == std::string::npos ? key : key.substr(pos + 1);
197
198 if (group != previousGroup) {
199 if (!previousGroup.empty())
200 os << "\n";
201 if (!group.empty())
202 os << group << ":\n";
203 previousGroup = group;
204 }
205
206 const std::string indent = group.empty() ? "" : " ";
207 os << indent << leaf << ": " << formatDefault(opt);
208 if (!opt.description.empty())
209 os << " # " << opt.description;
210 if (opt.type == OptionType::Enum) {
211 os << " [";
212 for (std::size_t i = 0; i < opt.enumValues.size(); ++i) {
213 if (i)
214 os << "|";
215 os << opt.enumValues[i];
216 }
217 os << "]";
218 } else {
219 os << " [" << typeName(opt.type) << "]";
220 }
221 os << "\n";
222 }
223}
