50std::unique_ptr<Solver> makeSaturnStokes(const Config& cfg) {
51 Mesh mesh = makeMeshForSolver(cfg);
52 if (mesh.dimension != 2 && mesh.dimension != 3)
53 throw ConfigError("saturnStokes supports structured 2D and 3D meshes");
54 if (mesh.nz < 1 || (mesh.dimension == 2 && mesh.nz != 1))
55 throw ConfigError("saturnStokes: 2D requires nz = 1");
56
57 StokesOptions options;
58 options.mu = cfg.getDouble("stokes.mu", 1.0);
59 options.stabilization = cfg.getDouble("stokes.stabilization", 1.0);
60 options.periodicX = true;
61 options.periodicZ = true;
62 options.periodicY = cfg.getBool("stokes.periodicY", false);
63 options.direct =
64 cfg.getString("stokes.linearSolver", "fieldsplit") == "direct";
65 options.velocityKsp = cfg.getString("stokes.velocityKsp", "gmres");
66 options.velocityPc = cfg.getString("stokes.velocityPc", "gamg");
67 options.pressureKsp = cfg.getString("stokes.pressureKsp", "preonly");
68 options.pressurePc = cfg.getString("stokes.pressurePc", "gamg");
69 options.force = {cfg.getDouble("stokes.forceX", 0.0),
70 cfg.getDouble("stokes.forceY", 0.0),
71 cfg.getDouble("stokes.forceZ", 0.0)};
72
73 const std::string test = cfg.getString("stokes.case", "poiseuille");
74 const Real y0 = mesh.bboxMin.y;
75 const Real H = mesh.bboxMax.y - y0;
76
77 if (test == "poiseuille") {
78 const Real G = cfg.getDouble("stokes.forceX", 1.0);
79 const Real mu = options.mu;
80 options.force = {G, 0, 0};
81 options.periodicY = false;
82 options.exactVelocity = [G, mu, y0, H](const Vec3& x) {
83 const Real yy = x.y - y0;
84 return Vec3{G / (2 * mu) * yy * (H - yy), 0, 0};
85 };
86 options.exactPressure = [](const Vec3&) { return Real(0); };
87 } else if (test == "couette") {
88 const Real U = cfg.getDouble("stokes.wallVelocityX", 1.0);
89 options.force = {0, 0, 0};
90 options.periodicY = false;
91 options.wallTopU = U;
92 options.exactVelocity = [U, y0, H](const Vec3& x) {
93 return Vec3{U * (x.y - y0) / H, 0, 0};
94 };
95 options.exactPressure = [](const Vec3&) { return Real(0); };
96 } else if (test == "mms") {
97 options.force = {0, 0, 0};
98 const MmsVariant variant = cfg.getBool("stokes.mmsWall", false)
99 ? MmsVariant::Wall
100 : MmsVariant::Periodic;
101 options.periodicY = (variant == MmsVariant::Periodic);
102 const int dim = mesh.dimension;
103 const Real hx =
104 (mesh.bboxMax.x - mesh.bboxMin.x) / static_cast<Real>(mesh.nx);
105 const Real hy =
106 (mesh.bboxMax.y - mesh.bboxMin.y) / static_cast<Real>(mesh.ny);
107 const Real hz = (dim == 3) ? (mesh.bboxMax.z - mesh.bboxMin.z) /
108 static_cast<Real>(mesh.nz)
109 : hy;
110 const Real hMin = std::min(hx, std::min(hy, hz));
111 const Real beta = options.stabilization * hMin * hMin / options.mu;
112 // The pressure reference row pins p = 0 at the first cell centre. Shift
113 // the manufactured pressure so that it is exactly zero there; a constant
114 // shift does not change grad(p) or lap(p), hence not f or g either.
115 const Vec3 pinPoint{mesh.bboxMin.x + Real(0.5) * hx,
116 mesh.bboxMin.y + Real(0.5) * hy,
117 (dim == 3 ? mesh.bboxMin.z + Real(0.5) * hz : Real(0))};
118 const Real pPin = mmsPressure(pinPoint, dim, variant);
119 options.bodyForce = [dim, variant, mu = options.mu](const Vec3& x) {
120 return mmsForce(x, dim, variant, mu);
121 };
122 options.continuitySource = [dim, variant, beta](const Vec3& x) {
123 return -beta * mmsPressureLaplacian(x, dim, variant);
124 };
125 options.exactVelocity = [dim, variant](const Vec3& x) {
126 return mmsVelocity(x, dim, variant);
127 };
128 options.exactPressure = [dim, variant, pPin](const Vec3& x) {
129 return mmsPressure(x, dim, variant) - pPin;
130 };
131 } else {
132 throw ConfigError("unsupported stokes.case '" + test + "'");
133 }
134
135 return std::make_unique<SaturnStokes>(std::move(mesh), std::move(options));
136}
