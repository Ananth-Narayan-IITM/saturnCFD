104void LocalMesh::build(const Mesh& mesh) {
105 const int nCells = static_cast<int>(mesh.cells.size());
106 std::vector<std::vector<int>> cellFaces(nCells);
107 for (int f = 0; f < static_cast<int>(mesh.faces.size()); ++f) {
108 const Face& face = mesh.faces[f];
109 if (face.owner >= 0)
110 cellFaces[face.owner].push_back(f);
111 if (face.neighbour >= 0)
112 cellFaces[face.neighbour].push_back(f);
113 }
114
115 mPatchNames.clear();
116 for (const Patch& patch : mesh.patches)
117 mPatchNames.push_back(patch.name);
118
119 for (int k = mHost.origin[2]; k < mHost.origin[2] + mHost.extent[2]; ++k) {
120 for (int j = mHost.origin[1]; j < mHost.origin[1] + mHost.extent[1]; ++j) {
121 for (int i = mHost.origin[0]; i < mHost.origin[0] + mHost.extent[0];
122 ++i) {
123 const int cell = mesh.cellAt(i, j, k);
124 if (cell < 0)
125 throw MeshError("LocalMesh: owned cell missing from the mesh");
126 const int ii = mHost.localIndex(i, 0);
127 const int jj = mHost.localIndex(j, 1);
128 const int kk = mHost.localIndex(k, 2);
129
130 mHost.mVolume(ii, jj, kk) = mesh.cellVolume[cell];
131 mHost.mCellId(ii, jj, kk) = cell;
132 const Vec3& centre = mesh.cellCentroid[cell];
133 mHost.mCentroid(ii, jj, kk, 0) = centre.x;
134 mHost.mCentroid(ii, jj, kk, 1) = centre.y;
135 mHost.mCentroid(ii, jj, kk, 2) = centre.z;
136
137 std::array<bool, 6> seen{};
138 for (int fi : cellFaces[cell]) {
139 const Face& face = mesh.faces[fi];
140 const bool isOwner = face.owner == cell;
141 if (!isOwner && face.neighbour != cell)
142 throw MeshError("LocalMesh: face is not adjacent to its cell");
143 if (face.neighbour < 0 && face.patch < 0)
144 throw MeshError("LocalMesh: boundary face has no patch");
145
146 const Vec3 outward = isOwner ? face.normal : -face.normal;
147 int axis = 0;
148 Real best = std::abs(outward[0]);
149 for (int a = 1; a < 3; ++a) {
150 const Real candidate = std::abs(outward[a]);
151 if (candidate > best) {
152 best = candidate;
153 axis = a;
154 }
155 }
156 if (best <= 0)
157 throw MeshError("LocalMesh: degenerate face normal");
158 const int dir = 2 * axis + (outward[axis] > 0 ? 1 : 0);
159 if (seen[dir])
160 throw MeshError("LocalMesh: duplicate face direction");
161 seen[dir] = true;
162
163 mHost.mFaceArea(ii, jj, kk, dir) = face.area;
164 mHost.mFaceNormal(ii, jj, kk, dir * 3 + 0) = outward.x;
165 mHost.mFaceNormal(ii, jj, kk, dir * 3 + 1) = outward.y;
166 mHost.mFaceNormal(ii, jj, kk, dir * 3 + 2) = outward.z;
167 mHost.mFaceCentroid(ii, jj, kk, dir * 3 + 0) = face.centroid.x;
168 mHost.mFaceCentroid(ii, jj, kk, dir * 3 + 1) = face.centroid.y;
169 mHost.mFaceCentroid(ii, jj, kk, dir * 3 + 2) = face.centroid.z;
170
171 const Real distance = dot(face.centroid - centre, outward);
172 if (!(distance > 0))
173 throw MeshError("LocalMesh: non-positive face distance");
174 mHost.mFaceDistance(ii, jj, kk, dir) = distance;
175 mHost.mFacePatch(ii, jj, kk, dir) =
176 (face.neighbour >= 0) ? -1 : face.patch;
177 }
178 for (int dir = 0; dir < 6; ++dir)
179 if (!seen[dir])
180 throw MeshError("LocalMesh: cell does not have six distinct faces");
181 }
182 }
183 }
184
185 // Ghost-cell centroids are needed by the non-orthogonal Laplacian on faces
186 // that touch a partition or periodic ghost. A physical-boundary ghost is
187 // handled by the mirror formula and does not need a centroid.
188 Vec3 period[3]{};
189 for (int d = 0; d < 3; ++d) {
190 if (!mHost.periodic[d] || mGlobal[d] <= 1)
191 continue;
192 int i1 = 0, j1 = 0, k1 = 0;
193 if (d == 0)
194 i1 = 1;
195 else if (d == 1)
196 j1 = 1;
197 else
198 k1 = 1;
199 const int c0 = mesh.cellAt(0, 0, 0);
200 const int c1 = mesh.cellAt(i1, j1, k1);
201 if (c0 < 0 || c1 < 0)
202 throw MeshError("LocalMesh: cannot determine the periodic lattice");
203 period[d] =
204 (mesh.cellCentroid[c1] - mesh.cellCentroid[c0]) * Real(mGlobal[d]);
205 }
206
207 const int ng = mHost.nGrow;
208 for (int k = mHost.origin[2] - ng; k < mHost.origin[2] + mHost.extent[2] + ng;
209 ++k) {
210 for (int j = mHost.origin[1] - ng;
211 j < mHost.origin[1] + mHost.extent[1] + ng; ++j) {
212 for (int i = mHost.origin[0] - ng;
213 i < mHost.origin[0] + mHost.extent[0] + ng; ++i) {
214 if (mHost.owned(i, j, k))
215 continue;
216 int wrapped[3] = {i, j, k};
217 Vec3 shift{};
218 bool valid = true;
219 for (int d = 0; d < 3; ++d) {
220 const int index = (d == 0) ? i : ((d == 1) ? j : k);
221 const int n = mGlobal[d];
222 if (index >= 0 && index < n)
223 continue;
224 if (!mHost.periodic[d]) {
225 valid = false;
226 break;
227 }
228 int q = index / n;
229 if (index < 0 && index % n != 0)
230 --q;
231 wrapped[d] = index - q * n;
232 shift += period[d] * Real(q);
233 }
234 if (!valid)
235 continue;
236 const int cell = mesh.cellAt(wrapped[0], wrapped[1], wrapped[2]);
237 if (cell < 0)
238 throw MeshError("LocalMesh: wrapped ghost cell is not in the mesh");
239 const Vec3 x = mesh.cellCentroid[cell] + shift;
240 const int ii = mHost.localIndex(i, 0);
241 const int jj = mHost.localIndex(j, 1);
242 const int kk = mHost.localIndex(k, 2);
243 mHost.mCentroid(ii, jj, kk, 0) = x.x;
244 mHost.mCentroid(ii, jj, kk, 1) = x.y;
245 mHost.mCentroid(ii, jj, kk, 2) = x.z;
246 }
247 }
248 }

251LocalMesh::Serialized LocalMesh::pack() const {
252 const int nxg = mHost.extent[0] + 2 * mHost.nGrow;
253 const int nyg = mHost.extent[1] + 2 * mHost.nGrow;
254 const int nzg = mHost.extent[2] + 2 * mHost.nGrow;
255 const int cells = nxg * nyg * nzg;
256 Serialized out;
257 out.reals.reserve(static_cast<std::size_t>(cells) * 52);
258 out.ints.reserve(static_cast<std::size_t>(cells) * 7);
259 for (int k = 0; k < nzg; ++k) {
260 for (int j = 0; j < nyg; ++j) {
261 for (int i = 0; i < nxg; ++i) {
262 out.reals.push_back(mHost.mVolume(i, j, k));
263 for (int c = 0; c < 3; ++c)
264 out.reals.push_back(mHost.mCentroid(i, j, k, c));
265 for (int c = 0; c < 6; ++c)
266 out.reals.push_back(mHost.mFaceArea(i, j, k, c));
267 for (int c = 0; c < 18; ++c)
268 out.reals.push_back(mHost.mFaceNormal(i, j, k, c));
269 for (int c = 0; c < 18; ++c)
270 out.reals.push_back(mHost.mFaceCentroid(i, j, k, c));
271 for (int c = 0; c < 6; ++c)
272 out.reals.push_back(mHost.mFaceDistance(i, j, k, c));
273 for (int c = 0; c < 6; ++c)
274 out.ints.push_back(mHost.mFacePatch(i, j, k, c));
275 out.ints.push_back(mHost.mCellId(i, j, k));
276 }
277 }
278 }
279 out.patches = mPatchNames;
280 return out;

283LocalMesh LocalMesh::unpack(const PartitionBox& box, int nGrow,
284 const Serialized& data) {
285 LocalMesh local;
286 const int dimension = (box.global[2] > 1) ? 3 : ((box.global[1] > 1) ? 2 : 1);
287 for (int d = 0; d < 3; ++d) {
288 local.mGlobal[d] = box.global[d];
289 local.mDevice.global[d] = box.global[d];
290 local.mHost.global[d] = box.global[d];
291 local.mDevice.origin[d] = box.origin[d];
292 local.mHost.origin[d] = box.origin[d];
293 local.mDevice.extent[d] = box.extent[d];
294 local.mHost.extent[d] = box.extent[d];
295 local.mDevice.periodic[d] = box.periodic[d];
296 local.mHost.periodic[d] = box.periodic[d];
297 }
298 local.mGlobalCellCount =
299 static_cast<BigIndex>(box.global[0]) * box.global[1] * box.global[2];
300 local.allocate(nGrow, dimension);
301 local.mPatchNames = data.patches;
302
303 const std::size_t expectedReals =
304 static_cast<std::size_t>(local.mHost.extent[0] + 2 * nGrow) *
305 (local.mHost.extent[1] + 2 * nGrow) *
306 (local.mHost.extent[2] + 2 * nGrow) * 52;
307 const std::size_t expectedInts =
308 static_cast<std::size_t>(local.mHost.extent[0] + 2 * nGrow) *
309 (local.mHost.extent[1] + 2 * nGrow) *
310 (local.mHost.extent[2] + 2 * nGrow) * 7;
311 if (data.reals.size() != expectedReals || data.ints.size() != expectedInts)
312 throw MeshError("LocalMesh::unpack: local geometry size mismatch");
313
314 std::size_t r = 0;
315 std::size_t q = 0;
316 const int nxg = local.mHost.extent[0] + 2 * nGrow;
317 const int nyg = local.mHost.extent[1] + 2 * nGrow;
318 const int nzg = local.mHost.extent[2] + 2 * nGrow;
319 for (int k = 0; k < nzg; ++k) {
320 for (int j = 0; j < nyg; ++j) {
321 for (int i = 0; i < nxg; ++i) {
322 local.mHost.mVolume(i, j, k) = data.reals[r++];
323 for (int c = 0; c < 3; ++c)
324 local.mHost.mCentroid(i, j, k, c) = data.reals[r++];
325 for (int c = 0; c < 6; ++c)
326 local.mHost.mFaceArea(i, j, k, c) = data.reals[r++];
327 for (int c = 0; c < 18; ++c)
328 local.mHost.mFaceNormal(i, j, k, c) = data.reals[r++];
329 for (int c = 0; c < 18; ++c)
330 local.mHost.mFaceCentroid(i, j, k, c) = data.reals[r++];
331 for (int c = 0; c < 6; ++c)
332 local.mHost.mFaceDistance(i, j, k, c) = data.reals[r++];
333 for (int c = 0; c < 6; ++c)
334 local.mHost.mFacePatch(i, j, k, c) = data.ints[q++];
335 local.mHost.mCellId(i, j, k) = data.ints[q++];
336 }
337 }
338 }
339
340 Kokkos::deep_copy(local.mDevice.mVolume, local.mHost.mVolume);
341 Kokkos::deep_copy(local.mDevice.mCentroid, local.mHost.mCentroid);
342 Kokkos::deep_copy(local.mDevice.mFaceArea, local.mHost.mFaceArea);
343 Kokkos::deep_copy(local.mDevice.mFaceNormal, local.mHost.mFaceNormal);
344 Kokkos::deep_copy(local.mDevice.mFaceCentroid, local.mHost.mFaceCentroid);
345 Kokkos::deep_copy(local.mDevice.mFaceDistance, local.mHost.mFaceDistance);
346 Kokkos::deep_copy(local.mDevice.mFacePatch, local.mHost.mFacePatch);
347 Kokkos::deep_copy(local.mDevice.mCellId, local.mHost.mCellId);
348 return local;

351LocalMesh LocalMesh::distributed(const Mesh& mesh, const MeshPartition& part,
352 int nGrow) {
353#ifndef SATURN_USE_MPI
354 return LocalMesh(mesh, part, nGrow);
355#else
356 int rank = 0;
357 int size = 1;
358 MPI_Comm_rank(MPI_COMM_WORLD, &rank);
359 MPI_Comm_size(MPI_COMM_WORLD, &size);
360 if (size == 1)
361 return LocalMesh(mesh, part.box(), nGrow);
362
363 if (rank == 0) {
364 LocalMesh local(mesh, part.box(), nGrow);
365 for (int target = 1; target < size; ++target) {
366 PartitionBox box = part.boxForRank(target);
367 LocalMesh temp(mesh, box, nGrow);
368 Serialized data = temp.pack();
369
370 std::string blob;
371 for (const std::string& name : data.patches) {
372 blob += name;
373 blob.push_back('\0');
374 }
375 int sizes[3] = {static_cast<int>(data.reals.size()),
376 static_cast<int>(data.ints.size()),
377 static_cast<int>(blob.size())};
378 MPI_Send(sizes, 3, MPI_INT, target, 700, MPI_COMM_WORLD);
379 if (!data.reals.empty())
380 MPI_Send(data.reals.data(), sizes[0], MPI_DOUBLE, target, 701,
381 MPI_COMM_WORLD);
382 if (!data.ints.empty())
383 MPI_Send(data.ints.data(), sizes[1], MPI_INT, target, 702,
384 MPI_COMM_WORLD);
385 if (!blob.empty())
386 MPI_Send(blob.data(), sizes[2], MPI_CHAR, target, 703, MPI_COMM_WORLD);
387 }
388 return local;
389 }
390
391 int sizes[3] = {0, 0, 0};
392 MPI_Recv(sizes, 3, MPI_INT, 0, 700, MPI_COMM_WORLD, MPI_STATUS_IGNORE);
393 Serialized data;
394 data.reals.resize(static_cast<std::size_t>(sizes[0]));
395 data.ints.resize(static_cast<std::size_t>(sizes[1]));
396 std::string blob(static_cast<std::size_t>(sizes[2]), '\0');
397 if (sizes[0] > 0)
398 MPI_Recv(data.reals.data(), sizes[0], MPI_DOUBLE, 0, 701, MPI_COMM_WORLD,
399 MPI_STATUS_IGNORE);
400 if (sizes[1] > 0)
401 MPI_Recv(data.ints.data(), sizes[1], MPI_INT, 0, 702, MPI_COMM_WORLD,
402 MPI_STATUS_IGNORE);
403 if (sizes[2] > 0)
404 MPI_Recv(blob.data(), sizes[2], MPI_CHAR, 0, 703, MPI_COMM_WORLD,
405 MPI_STATUS_IGNORE);
406
407 std::size_t start = 0;
408 while (start < blob.size()) {
409 const std::size_t end = blob.find('\0', start);
410 if (end == std::string::npos)
411 break;
412 data.patches.emplace_back(blob.substr(start, end - start));
413 start = end + 1;
414 }
415 return unpack(part.box(), nGrow, data);
416#endif
417}
