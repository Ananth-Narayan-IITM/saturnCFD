18std::string safeName(const std::string& name) {
19 std::string out;
20 for (unsigned char c : name) {
21 if (std::isalnum(c))
22 out += static_cast<char>(c);
23 }
24 if (out.empty())
25 out = "patch";
26 if (std::isdigit(static_cast<unsigned char>(out.front())))
27 out = "patch" + out;
28 return out;
29}

144

150 for (Index fi : patch.faces) {

153 for (int i = 0; i < face.nNodes; ++i) {

163 }
