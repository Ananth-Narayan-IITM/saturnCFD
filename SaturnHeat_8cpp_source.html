33SaturnHeat::SaturnHeat(Mesh mesh, HeatOptions options)
34 : mMesh(std::move(mesh)),
35 mOptions(std::move(options)),
36 mPart(mMesh.nx, mMesh.ny, mMesh.nz, mOptions.periodic && mMesh.nx > 1,
37 mOptions.periodic && mMesh.dimension >= 2 && mMesh.ny > 1,
38 mOptions.periodic && mMesh.dimension >= 3 && mMesh.nz > 1),
39 mLocal(LocalMesh::distributed(mMesh, mPart, 2)),
40 mHalo(mPart, 2, 1),
41 mFieldHalo(mHalo, 1),
42 mT(mPart, 2, 1, "saturnHeat::T"),
43 mFlux(mPart, 2, 1, "saturnHeat::flux"),
44 mNormalVelocity(mPart, 2, 1, "saturnHeat::normal_velocity"),
45 mLapFace(mPart, 2, 1, "saturnHeat::lap_face"),
46 mDiv(mPart, 2, 1, "saturnHeat::div"),
47 mLap(mPart, 2, 1, "saturnHeat::lap"),
48 mGrad(mPart, 2, 3, "saturnHeat::grad") {
49 if (!mMesh.structured || mMesh.dimension < 1 || mMesh.dimension > 3)
50 throw std::invalid_argument(
51 "saturnHeat: expected a structured 1D, 2D or 3D mesh");
52 if (mMesh.dimension == 1 && (mMesh.ny != 1 || mMesh.nz != 1))
53 throw std::invalid_argument("saturnHeat: 1D requires ny = nz = 1");
54 if (mMesh.dimension == 2 && mMesh.nz != 1)
55 throw std::invalid_argument("saturnHeat: 2D requires nz = 1");
56 if (mMesh.nx < 1 || mMesh.ny < 1 || mMesh.nz < 1)
57 throw std::invalid_argument("saturnHeat: empty mesh");
58 if (mOptions.alpha < 0)
59 throw std::invalid_argument("saturnHeat: negative diffusivity");
60 if (!mOptions.initial)
61 throw std::invalid_argument("saturnHeat: initial condition is missing");
62 if (!mOptions.periodic && !mOptions.dirichlet)
63 throw std::invalid_argument(
64 "saturnHeat: non-periodic boundaries need a Dirichlet callback");
65
66 Real minH = std::numeric_limits<Real>::max();
67 const Real exponent = Real(1) / static_cast<Real>(mMesh.dimension);
68 for (int k = mPart.localOrigin(2);
69 k < mPart.localOrigin(2) + mPart.localExtent(2); ++k) {
70 for (int j = mPart.localOrigin(1);
71 j < mPart.localOrigin(1) + mPart.localExtent(1); ++j) {
72 for (int i = mPart.localOrigin(0);
73 i < mPart.localOrigin(0) + mPart.localExtent(0); ++i)
74 minH = std::min(minH, std::pow(mLocal.cellVolume(i, j, k), exponent));
75 }
76 }
77 if (!(minH > 0))
78 throw std::invalid_argument("saturnHeat: non-positive cell size");
79 mMinCellSize = minH;
80}

82void SaturnHeat::initialize() {
83 FaceField un = mNormalVelocity;
84 const LocalMesh::DeviceData mesh = mLocal.device();
85 const Vec3 velocity = mOptions.velocity;
86 const int i0 = mesh.origin[0];
87 const int j0 = mesh.origin[1];
88 const int k0 = mesh.origin[2];
89 Kokkos::parallel_for(
90 "saturnHeat::velocity_faces",
91 Policy3d({i0, j0, k0},
92 {i0 + mesh.extent[0], j0 + mesh.extent[1], k0 + mesh.extent[2]}),
93 KOKKOS_LAMBDA(int i, int j, int k) {
94 for (int dir = 0; dir < 2 * mesh.dimension; ++dir)
95 un(i, j, k, dir, 0) = dot(velocity, mesh.faceNormal(i, j, k, dir));
96 });
97}

100 auto host = mT.hostView();
101 for (int k = mPart.localOrigin(2);
102 k < mPart.localOrigin(2) + mPart.localExtent(2); ++k) {
103 for (int j = mPart.localOrigin(1);
104 j < mPart.localOrigin(1) + mPart.localExtent(1); ++j) {

109 const int kk = k - (mT.origin(2) - mT.nGrow());

111 }

134void SaturnHeat::rhs(const CellField& state, CellField& rate) {
135 if (state.ncomp() != 1)
136 throw std::invalid_argument("saturnHeat: expected a scalar state");
137 convectiveFlux(mNormalVelocity, state, mFlux, mLocal, mOptions.convection,
138 mOptions.limiter);
139 divergence(mFlux, mDiv, mLocal);
140 laplacian(state, mLapFace, mGrad, mLap, mLocal);
141
142 CellField r = rate;
143 CellField div = mDiv;
144 CellField lap = mLap;
145 const Real alpha = mOptions.alpha;
146 const Real source = mOptions.source;
147 Kokkos::parallel_for(
148 "saturnHeat::rhs", ownedPolicy(state),
149 KOKKOS_LAMBDA(int i, int j, int k) {
150 r(i, j, k, 0) = -div(i, j, k, 0) + alpha * lap(i, j, k, 0) + source;
151 });
152}

162template <class FieldAccess>
163std::vector<ScalarProfilePoint> gatherProfile(
164 const LocalMesh& local, const MeshPartition& part, int i0, int j0, int k0,
165 int nx, int ny, int nz, FieldAccess value, Real time,
166 const std::function<Real(const Vec3&, Real)>& exact, bool withExact,
167 BigIndex globalCellCount) {
168 const int n = nx * ny * nz;
169 std::vector<double> packed(static_cast<std::size_t>(6) * n);
170 int a = 0;
171 for (int k = k0; k < k0 + nz; ++k) {
172 for (int j = j0; j < j0 + ny; ++j) {
173 for (int i = i0; i < i0 + nx; ++i) {
174 const int cell = local.cellId(i, j, k);
175 if (cell < 0)
176 throw std::runtime_error("saturnHeat: missing owned cell");
177 const Vec3 centre = local.cellCentroid(i, j, k);
178 const std::size_t base = static_cast<std::size_t>(6) * a;
179 packed[base + 0] = centre.x;
180 packed[base + 1] = centre.y;
181 packed[base + 2] = centre.z;
182 packed[base + 3] = value(i, j, k);
183 packed[base + 4] = withExact ? exact(centre, time) : 0.0;
184 packed[base + 5] = static_cast<double>(cell);
185 ++a;
186 }
187 }
188 }
189
190 std::vector<double> global;
191#ifdef SATURN_USE_MPI
192 int localCount = n;
193 int worldSize = 1;
194 MPI_Comm_size(MPI_COMM_WORLD, &worldSize);
195 std::vector<int> counts(static_cast<std::size_t>(worldSize));
196 std::vector<int> offsets(static_cast<std::size_t>(worldSize));
197 MPI_Allgather(&localCount, 1, MPI_INT, counts.data(), 1, MPI_INT,
198 MPI_COMM_WORLD);
199 offsets[0] = 0;
200 for (int r = 1; r < worldSize; ++r)
201 offsets[static_cast<std::size_t>(r)] =
202 offsets[static_cast<std::size_t>(r - 1)] +
203 counts[static_cast<std::size_t>(r - 1)];
204 const int total = offsets.back() + counts.back();
205 std::vector<int> doubleCounts(static_cast<std::size_t>(worldSize));
206 std::vector<int> doubleOffsets(static_cast<std::size_t>(worldSize));
207 for (int r = 0; r < worldSize; ++r) {
208 doubleCounts[static_cast<std::size_t>(r)] =
209 6 * counts[static_cast<std::size_t>(r)];
210 doubleOffsets[static_cast<std::size_t>(r)] =
211 6 * offsets[static_cast<std::size_t>(r)];
212 }
213 global.resize(static_cast<std::size_t>(6) * total);
214 MPI_Allgatherv(packed.data(), 6 * n, MPI_DOUBLE, global.data(),
215 doubleCounts.data(), doubleOffsets.data(), MPI_DOUBLE,
216 MPI_COMM_WORLD);
217#else
218 global = std::move(packed);
219#endif
220
221 const std::size_t pointCount = global.size() / 6;
222 std::vector<ScalarProfilePoint> points(pointCount);
223 for (std::size_t p = 0; p < pointCount; ++p) {
224 const std::size_t base = 6 * p;
225 points[p].cell = static_cast<Index>(global[base + 5]);
226 points[p].x = global[base + 0];
227 points[p].y = global[base + 1];
228 points[p].z = global[base + 2];
229 points[p].numerical = global[base + 3];
230 points[p].exact = global[base + 4];
231 points[p].error = points[p].numerical - points[p].exact;
232 }
233 std::sort(points.begin(), points.end(),
234 [](const ScalarProfilePoint& a, const ScalarProfilePoint& b) {
235 if (a.x != b.x)
236 return a.x < b.x;
237 if (a.y != b.y)
238 return a.y < b.y;
239 return a.z < b.z;
240 });
241 (void)part;
242 (void)globalCellCount;
243 return points;
244}

248std::vector<ScalarProfilePoint> SaturnHeat::profile(Real time,
249 bool withExact) const {
250 if (withExact && !mOptions.exact)
251 throw std::runtime_error(
252 "saturnHeat: profile comparison needs an exact solution");

261 mPart.localOrigin(1), mPart.localOrigin(2),
262 mPart.localExtent(0), mPart.localExtent(1),
263 mPart.localExtent(2), value, time, mOptions.exact,
264 withExact, mLocal.globalCellCount());

267ScalarError SaturnHeat::exactError(Real time) const {
268 const std::vector<ScalarProfilePoint> points = profile(time);
269 ScalarError out;
270 out.samples = static_cast<BigIndex>(points.size());
271 if (points.empty())
272 return out;
273 double l1 = 0;
274 double l2 = 0;
275 double linf = 0;
276 for (const ScalarProfilePoint& p : points) {
277 const double e = static_cast<double>(p.error);
278 l1 += std::abs(e);
279 l2 += e * e;
280 linf = std::max(linf, std::abs(e));
281 }
282 out.l1 = static_cast<Real>(l1 / static_cast<double>(points.size()));
283 out.l2 =
284 static_cast<Real>(std::sqrt(l2 / static_cast<double>(points.size())));
285 out.linf = static_cast<Real>(linf);
286 return out;
287}
