35Vec3 mmsForce(const Vec3& x, int dim, MmsVariant variant, Real mu) {
36 const Real k = 2 * kPi;
37 const Real k2 = k * k;
38 const Vec3 u = mmsVelocity(x, dim, variant);

41 Vec3 f;

43 k * std::cos(k * x.x) * std::sin(k * x.y) *
44 ((dim == 3) ? std::sin(k * x.z) : Real(1));
45 f.y = lapFactor * mu * k2 * u.y +
46 k * std::sin(k * x.x) * std::cos(k * x.y) *
47 ((dim == 3) ? std::sin(k * x.z) : Real(1));
48 f.z = (dim == 3)

54 // Wall variant: u is a product of sines in every direction, so
55 // lap(u) = -dim k^2 u. The v component is a product of sin^2(pi y),
56 // cos(k x) and sin(k z); its Laplacian is worked out analytically below.
57 const Real zs = (dim == 3) ? std::sin(k * x.z) : Real(1);
58 const Real zc = (dim == 3) ? std::cos(k * x.z) : Real(1);
59 const Real cx = std::cos(k * x.x);
60 const Real sx = std::sin(k * x.x);
61 const Real spy = std::sin(kPi * x.y);
62 const Real cy = std::cos(k * x.y);
63
64 const Real lapU = -((dim == 3) ? Real(3) : Real(2)) * k2 * u.x;
65 const Real lapV = (dim == 3) ? kPi * k2 * (2 - 3 * cy) * cx * zs
66 : kPi * k2 * (1 - 2 * cy) * cx;
67
68 Vec3 f;
69 f.x = -mu * lapU - k * sx * std::cos(kPi * x.y) * zc;
70 f.y = -mu * lapV - kPi * cx * spy * zc;
71 f.z =
72 (dim == 3) ? -k * cx * std::cos(kPi * x.y) * std::sin(k * x.z) : Real(0);
73 return f;
74}
