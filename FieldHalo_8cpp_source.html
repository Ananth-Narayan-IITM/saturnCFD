37 auto allocate = [this](const std::vector<HaloExchange::Op>& ops,
38 const char* label) {
39 std::vector<Buffer> out;
40 out.reserve(ops.size());
41 for (const auto& op : ops) {
42 Buffer b;
43 b.n = checkedSize(op.box.numPts() * mNcomp, label);

45 static_cast<std::size_t>(b.n));
46 b.host = Kokkos::create_mirror_view(b.dev);

56void FieldHalo::pack(int sendIndex, const CellField& field) {
57 const HaloExchange::Op& op = mHalo.sendOps()[sendIndex];
58 Buffer& buffer = mSend[sendIndex];
59 const IndexBox box = op.box;
60 const int nx = box.length(0);
61 const int ny = box.length(1);
62 const int nz = box.length(2);
63 const int nc = mNcomp;
64 Kokkos::View<Real*> dev = buffer.dev;
65 CellField f = field;
66
67 Kokkos::parallel_for(
68 "saturn::halo_pack",
69 Kokkos::MDRangePolicy<Kokkos::Rank<3>, Kokkos::IndexType<int>>(
70 {0, 0, 0}, {nx, ny, nz}),
71 KOKKOS_LAMBDA(int ii, int jj, int kk) {
72 for (int c = 0; c < nc; ++c) {
73 const std::size_t t =
74 (static_cast<std::size_t>((kk * ny + jj) * nx + ii) *
75 static_cast<std::size_t>(nc)) +
76 static_cast<std::size_t>(c);
77 dev[t] = f(box.lo[0] + ii, box.lo[1] + jj, box.lo[2] + kk, c);
78 }
79 });
80
81 Kokkos::fence();
82 Kokkos::deep_copy(buffer.host, buffer.dev);
83}

136void FieldHalo::execute(CellField& field) {
137 if (field.ncomp() != mNcomp)
138 throw std::invalid_argument("FieldHalo: component count does not match");
139 if (field.nGrow() != mHalo.nGrow())
140 throw std::invalid_argument("FieldHalo: ghost width does not match");
141 for (int d = 0; d < 3; ++d) {
142 if (field.origin(d) != mHalo.partition().localOrigin(d) ||
143 field.extent(d) != mHalo.partition().localExtent(d))
144 throw std::invalid_argument(
145 "FieldHalo: field layout does not match the partition");
146 }
147
148 // Local periodic images (also the whole periodic path on one rank).
149 for (int r = 0; r < static_cast<int>(mRecv.size()); ++r)
150 if (mHalo.recvOps()[r].peer == mHalo.partition().rank())
151 copyLocal(r, field);
152
153#ifdef SATURN_USE_MPI
154 std::vector<MPI_Request> requests;
155 requests.reserve(mSend.size() + mRecv.size());
156
157 for (int r = 0; r < static_cast<int>(mRecv.size()); ++r) {
158 const HaloExchange::Op& op = mHalo.recvOps()[r];
159 if (op.peer == mHalo.partition().rank())
160 continue;
161 MPI_Request req = MPI_REQUEST_NULL;
162 MPI_Irecv(mRecv[r].host.data(), mRecv[r].n, kMpiReal, op.peer, op.tag,
163 MPI_COMM_WORLD, &req);
164 requests.push_back(req);
165 }
166
167 for (int s = 0; s < static_cast<int>(mSend.size()); ++s) {
168 const HaloExchange::Op& op = mHalo.sendOps()[s];
169 if (op.peer == mHalo.partition().rank())
170 continue;
171 pack(s, field);
172 MPI_Request req = MPI_REQUEST_NULL;
173 MPI_Isend(mSend[s].host.data(), mSend[s].n, kMpiReal, op.peer, op.tag,
174 MPI_COMM_WORLD, &req);
175 requests.push_back(req);
176 }
177
178 if (!requests.empty())
179 MPI_Waitall(static_cast<int>(requests.size()), requests.data(),
180 MPI_STATUSES_IGNORE);
181
182 for (int r = 0; r < static_cast<int>(mRecv.size()); ++r) {
183 const HaloExchange::Op& op = mHalo.recvOps()[r];
184 if (op.peer == mHalo.partition().rank())
185 continue;
186 unpack(r, field);
187 }
188#else
189 (void)field; // only local periodic copies exist without MPI
190#endif
191
192 Kokkos::fence();
193}
