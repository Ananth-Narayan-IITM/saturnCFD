61MeshReport checkMesh(Mesh& mesh, const std::string& file) {
62 MeshReport r;
63 r.file = file;
64
65 computeGeometry(mesh);
66 detectStructured(mesh);
67
68 r.nCells = static_cast<int>(mesh.cells.size());
69 r.nPoints = static_cast<int>(mesh.points.size());
70 r.nFaces = static_cast<int>(mesh.faces.size());
71 for (const Face& f : mesh.faces) {
72 if (f.neighbour >= 0)
73 ++r.nInternalFaces;
74 else
75 ++r.nBoundaryFaces;
76 }
77 for (const Patch& p : mesh.patches)
78 r.patches.emplace_back(p.name, static_cast<int>(p.faces.size()));
79
80 r.bboxMin = mesh.bboxMin;
81 r.bboxMax = mesh.bboxMax;
82 r.structured = mesh.structured;
83 r.dimension = mesh.dimension;
84 r.nx = mesh.nx;
85 r.ny = mesh.ny;
86 r.nz = mesh.nz;
87
88 if (!mesh.cellVolume.empty()) {
89 r.volumeMin =
90 *std::min_element(mesh.cellVolume.begin(), mesh.cellVolume.end());
91 r.volumeMax =
92 *std::max_element(mesh.cellVolume.begin(), mesh.cellVolume.end());
93 r.volumeTotal = std::accumulate(mesh.cellVolume.begin(),
94 mesh.cellVolume.end(), Real(0));
95 }
96
97 bool haveArea = false;
98 for (const Face& f : mesh.faces) {
99 if (f.area <= 0)
100 continue;
101 if (!haveArea) {
102 r.areaMin = r.areaMax = f.area;
103 haveArea = true;
104 } else {
105 r.areaMin = std::min(r.areaMin, f.area);
106 r.areaMax = std::max(r.areaMax, f.area);
107 }
108 }
109
110 r.nonOrthoMax = maximum(mesh.cellNonOrthogonality);
111 r.nonOrthoAvg = average(mesh.cellNonOrthogonality);
112 r.skewnessMax = maximum(mesh.cellSkewness);
113 r.skewnessAvg = average(mesh.cellSkewness);
114 r.aspectRatioMax = maximum(mesh.cellAspectRatio);
115 r.aspectRatioAvg = average(mesh.cellAspectRatio);
116
117 if (mesh.cells.empty())
118 r.errors.push_back("mesh has no volume cells");
119 for (const Cell& c : mesh.cells) {
120 if (c.type != 5)
121 r.errors.push_back("unsupported element type " + std::to_string(c.type) +
122 "; v1.0 accepts hexahedra only");
123 }
124 for (int i = 0; i < static_cast<int>(mesh.cellVolume.size()); ++i) {
125 if (mesh.cellVolume[i] <= 0)
126 r.errors.push_back("cell " + std::to_string(i) +
127 " has non-positive volume");
128 }
129 for (int i = 0; i < static_cast<int>(mesh.faces.size()); ++i) {
130 const Face& f = mesh.faces[i];
131 if (f.area <= 0)
132 r.errors.push_back("face " + std::to_string(i) + " has zero area");
133 if (f.neighbour < 0 && f.patch < 0)
134 r.errors.push_back("boundary face " + std::to_string(i) +
135 " is not assigned to a patch");
136 }
137 if (!mesh.structured)
138 r.errors.push_back(
139 "mesh is not structured; only contiguous structured hexahedral "
140 "meshes are supported in v1.0");
141
142 if (r.nonOrthoMax > 60)
143 r.warnings.push_back("maximum non-orthogonality exceeds 60 degrees");
144 if (r.skewnessMax > 0.5)
145 r.warnings.push_back("maximum skewness exceeds 0.5");
146 if (r.aspectRatioMax > 100)
147 r.warnings.push_back("maximum aspect ratio exceeds 100");
148 if (r.volumeMin > 0 && r.volumeMax / r.volumeMin > 1.0e4)
149 r.warnings.push_back("cell volume ratio exceeds 1e4");
150
151 return r;
152}
