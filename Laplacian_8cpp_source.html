9void laplacian(const CellField& phi, CellField& out,
10 const CartesianGeometry& geom) {
11 if (phi.ncomp() != out.ncomp())
12 throw std::invalid_argument("laplacian: component count mismatch");
13 for (int d = 0; d < 3; ++d) {
14 if (phi.origin(d) != out.origin(d) || phi.extent(d) != out.extent(d))
15 throw std::invalid_argument("laplacian: field layout mismatch");
16 }
17
18 const int i0 = phi.origin(0);
19 const int j0 = phi.origin(1);
20 const int k0 = phi.origin(2);
21 const int nx = phi.extent(0);
22 const int ny = phi.extent(1);
23 const int nz = phi.extent(2);
24 const int nc = phi.ncomp();
25 const int dim = geom.dim;
26 const CartesianGeometry g = geom;
27 CellField p = phi;
28 CellField o = out;
29
30 Kokkos::parallel_for(
31 "saturn::laplacian",
32 Kokkos::MDRangePolicy<Kokkos::Rank<3>, Kokkos::IndexType<int>>(
33 {i0, j0, k0}, {i0 + nx, j0 + ny, k0 + nz}),
34 KOKKOS_LAMBDA(int i, int j, int k) {
35 for (int c = 0; c < nc; ++c) {
36 Real sum = 0;
37 for (int d = 0; d < 3; ++d) {
38 if (d >= dim)
39 break;
40 const int di = (d == 0) ? 1 : 0;
41 const int dj = (d == 1) ? 1 : 0;
42 const int dk = (d == 2) ? 1 : 0;
43 sum += g.weight(d) *
44 (p(i + di, j + dj, k + dk, c) +
45 p(i - di, j - dj, k - dk, c) - Real(2) * p(i, j, k, c));
46 }
47 o(i, j, k, c) = sum;
48 }
49 });

93 KOKKOS_LAMBDA(int i, int j, int k) {
94 Real sum = 0;
95 for (int dir = 0; dir < 2 * m.dimension; ++dir) {
96 int di = 0, dj = 0, dk = 0;
97 faceDirectionOffset(dir, di, dj, dk);
98 const int ni = i + di;
99 const int nj = j + dj;
100 const int nk = k + dk;
101 const Vec3 n = m.faceNormal(i, j, k, dir);
102 const Real area = m.faceArea(i, j, k, dir);
103 const Real phiP = p(i, j, k, 0);
104 const Real phiN = p(ni, nj, nk, 0);
105
106 Real flux = 0;
107 const bool physicalBoundary =
108 (m.boundaryPatch(i, j, k, dir) >= 0) && !m.periodic[dir / 2];
109 if (physicalBoundary) {
110 // Use the interpolated boundary-face value and correct for the
111 // tangential offset between the owner centroid and the face
112 // centroid. On an orthogonal mesh t is zero.
113 const Vec3 xP = m.cellCentroid(i, j, k);
114 const Vec3 xf = m.faceCentroid(i, j, k, dir);
115 const Vec3 dvec = xf - xP;
116 const Real dPerp = dot(dvec, n);
117 const Vec3 t = dvec - n * dPerp;
118 const Vec3 gradP = {g(i, j, k, 0), g(i, j, k, 1), g(i, j, k, 2)};
119 const Real denom =
120 (dPerp > 0) ? dPerp : m.faceDistance(i, j, k, dir);
121 flux = area * ((f(i, j, k, dir, 0) - phiP) - dot(t, gradP)) / denom;
122 } else {
123 const Vec3 xP = m.cellCentroid(i, j, k);
124 const Vec3 xN = m.cellCentroid(ni, nj, nk);
125 const Vec3 dvec = xN - xP;
126 const Real dPerp = dot(dvec, n);
127
128 Vec3 gradF{};
129 if (m.owned(ni, nj, nk)) {
130 const Vec3 xf = m.faceCentroid(i, j, k, dir);
131 const Vec3 dPv = xf - xP;
132 const Vec3 dNv = xf - xN;
133 const Real dP = Kokkos::sqrt(dot(dPv, dPv));
134 const Real dN = Kokkos::sqrt(dot(dNv, dNv));
135 const Real total = dP + dN;
136 const Real wP = (total > 0) ? dN / total : Real(0.5);
137 const Real wN = (total > 0) ? dP / total : Real(0.5);
138 gradF = {wP * g(i, j, k, 0) + wN * g(ni, nj, nk, 0),
139 wP * g(i, j, k, 1) + wN * g(ni, nj, nk, 1),
140 wP * g(i, j, k, 2) + wN * g(ni, nj, nk, 2)};
141 } else {
142 // Partition/periodic ghost: the owner gradient is the best
143 // available face gradient. It is exact for a linear field.
144 gradF = {g(i, j, k, 0), g(i, j, k, 1), g(i, j, k, 2)};
145 }
146
147 const Vec3 gradTang = gradF - n * dot(gradF, n);
148 const Real denom =
149 (dPerp > 0) ? dPerp : Real(2) * m.faceDistance(i, j, k, dir);
150 flux = area * ((phiN - phiP) - dot(dvec, gradTang)) / denom;
151 }
152 sum += flux;
153 }
154 o(i, j, k, 0) = sum / m.cellVolume(i, j, k);
155 });
