28void interpolate(const CellField& phi, FaceField& face, const LocalMesh& mesh) {
29 requireOwnedLayout(phi, mesh);
30 requireOwnedLayout(face, mesh);
31 if (phi.ncomp() != face.ncomp())
32 throw std::invalid_argument("interpolate: component count mismatch");
33
34 const LocalMesh::DeviceData m = mesh.device();
35 const int i0 = mesh.origin(0);
36 const int j0 = mesh.origin(1);
37 const int k0 = mesh.origin(2);
38 const int nx = mesh.extent(0);
39 const int ny = mesh.extent(1);
40 const int nz = mesh.extent(2);
41 const int nc = phi.ncomp();
42 CellField p = phi;
43 FaceField f = face;
44
45 Kokkos::parallel_for(
46 "saturn::interpolate",
47 Kokkos::MDRangePolicy<Kokkos::Rank<3>, Kokkos::IndexType<int>>(
48 {i0, j0, k0}, {i0 + nx, j0 + ny, k0 + nz}),
49 KOKKOS_LAMBDA(int i, int j, int k) {
50 for (int dir = 0; dir < 2 * m.dimension; ++dir) {
51 int di = 0, dj = 0, dk = 0;
52 faceDirectionOffset(dir, di, dj, dk);
53 const int ni = i + di;
54 const int nj = j + dj;
55 const int nk = k + dk;
56
57 Real wP = Real(0.5);
58 Real wN = Real(0.5);
59 const bool periodicFace = m.periodic[dir / 2];
60 const bool internalFace =
61 (m.boundaryPatch(i, j, k, dir) < 0) || periodicFace;
62 if (internalFace) {
63 const Vec3 xf = m.faceCentroid(i, j, k, dir);
64 const Vec3 xP = m.cellCentroid(i, j, k);
65 const Vec3 xN = m.cellCentroid(ni, nj, nk);
66 const Vec3 dPv = xf - xP;
67 const Vec3 dNv = xf - xN;
68 const Real dP = Kokkos::sqrt(dot(dPv, dPv));
69 const Real dN = Kokkos::sqrt(dot(dNv, dNv));
70 const Real total = dP + dN;
71 if (total > 0) {
72 wP = dN / total;
73 wN = dP / total;
74 }
75 }
76
77 for (int c = 0; c < nc; ++c)
78 f(i, j, k, dir, c) = wP * p(i, j, k, c) + wN * p(ni, nj, nk, c);
79 }
