20std::vector<std::string> split(const std::string& key) {
21 std::vector<std::string> parts;
22 std::string current;
23 for (char c : key) {
24 if (c == '.') {
25 if (current.empty())
26 throw ConfigError("invalid empty component in key '" + key + "'");
27 parts.push_back(current);
28 current.clear();
29 } else {
30 current += c;
31 }
32 }
33 if (current.empty())
34 throw ConfigError("invalid trailing dot in key '" + key + "'");
35 parts.push_back(current);
36 return parts;
37}

39void collectLeaves(const YAML::Node& node, const std::string& prefix,
40 std::vector<std::string>& out) {
41 if (!node || !node.IsMap()) {
42 if (!prefix.empty())
43 out.push_back(prefix);
44 return;
45 }
46 for (auto it = node.begin(); it != node.end(); ++it) {
47 const std::string key = it->first.as<std::string>();
48 const std::string path = prefix.empty() ? key : prefix + "." + key;
49 if (it->second.IsMap()) {
50 collectLeaves(it->second, path, out);
51 } else {
52 out.push_back(path);
53 }
54 }
55}
