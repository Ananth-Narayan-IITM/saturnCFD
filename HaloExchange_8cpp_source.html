34void HaloExchange::build() {
35 for (int dz = -1; dz <= 1; ++dz) {
36 for (int dy = -1; dy <= 1; ++dy) {
37 for (int dx = -1; dx <= 1; ++dx) {
38 if (dx == 0 && dy == 0 && dz == 0)
39 continue;
40 const int dir[3] = {dx, dy, dz};
41 bool touchesEmpty = false;
42 for (int d = 0; d < 3; ++d)
43 if (dir[d] != 0 && mPart.globalExtent(d) == 1)
44 touchesEmpty = true;
45 if (touchesEmpty)
46 continue;
47
48 const int sendPeer = mPart.neighbor(dx, dy, dz);
49 if (sendPeer != kProcNull) {
50 Op op;
51 op.peer = sendPeer;
52 op.dx = dx;
53 op.dy = dy;
54 op.dz = dz;
55 op.box = mPart.validSlab(mNGrow, dx, dy, dz);
56 op.tag = 1000 + dirIndex(dx, dy, dz);
57 mSendOps.push_back(std::move(op));
58 }
59
60 const int recvPeer = mPart.neighbor(dx, dy, dz);
61 if (recvPeer != kProcNull) {
62 Op op;
63 op.peer = recvPeer;
64 op.dx = dx;
65 op.dy = dy;
66 op.dz = dz;
67 op.box = mPart.ghostBox(mNGrow, dx, dy, dz);
68 op.tag = 1000 + dirIndex(-dx, -dy, -dz);
69 mRecvOps.push_back(std::move(op));
70 }
71 }
72 }

158void HaloExchange::execute(Real* data) {
159 // Same-rank periodic copies (and small process grids where a neighbour is
160 // the local rank) first.
161 for (const Op& r : mRecvOps) {
162 if (r.peer != mPart.rank())
163 continue;
164 const IndexBox src = mPart.validSlab(mNGrow, -r.dx, -r.dy, -r.dz);
165 copyBox(src, r.box, data);
166 }
167
168#ifdef SATURN_USE_MPI
169 std::vector<MPI_Request> requests;
170 requests.reserve(mSendOps.size() + mRecvOps.size());
171
172 for (Op& r : mRecvOps) {
173 if (r.peer == mPart.rank() || r.peer == kProcNull)
174 continue;
175 const BigIndex n = r.box.numPts() * mNcomp;
176 if (n > INT_MAX)
177 throw std::runtime_error("HaloExchange: message exceeds INT_MAX");
178 r.buffer.resize(static_cast<std::size_t>(n));
179 MPI_Request req = MPI_REQUEST_NULL;
180 MPI_Irecv(r.buffer.data(), static_cast<int>(n), MPI_DOUBLE, r.peer, r.tag,
181 MPI_COMM_WORLD, &req);
182 requests.push_back(req);
183 }
184
185 for (Op& s : mSendOps) {
186 if (s.peer == mPart.rank() || s.peer == kProcNull)
187 continue;
188 pack(s.box, data, s.buffer);
189 const BigIndex n = s.box.numPts() * mNcomp;
190 if (n > INT_MAX)
191 throw std::runtime_error("HaloExchange: message exceeds INT_MAX");
192 MPI_Request req = MPI_REQUEST_NULL;
193 MPI_Isend(s.buffer.data(), static_cast<int>(n), MPI_DOUBLE, s.peer, s.tag,
194 MPI_COMM_WORLD, &req);
195 requests.push_back(req);
196 }
197
198 if (!requests.empty())
199 MPI_Waitall(static_cast<int>(requests.size()), requests.data(),
200 MPI_STATUSES_IGNORE);
201
202 for (const Op& r : mRecvOps) {
203 if (r.peer == mPart.rank() || r.peer == kProcNull)
204 continue;
205 unpack(r.box, r.buffer, data);
206 }
207#else
208 (void)data;
209#endif
210}
