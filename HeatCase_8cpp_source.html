43std::unique_ptr<Solver> makeSaturnHeat(const Config& cfg) {
44 Mesh mesh = makeMeshForSolver(cfg);
45 if (mesh.dimension < 1 || mesh.dimension > 3)
46 throw ConfigError("saturnHeat supports structured 1D, 2D and 3D meshes");
47
48 HeatOptions options;
49 options.alpha = cfg.getDouble("heat.alpha", 0.01);
50 options.velocity = {cfg.getDouble("heat.velocityX", 1.0),
51 cfg.getDouble("heat.velocityY", 0.0),
52 cfg.getDouble("heat.velocityZ", 0.0)};
53 options.source = cfg.getDouble("heat.source", 0.0);
54 options.periodic = cfg.getBool("heat.periodic", false);
55 options.convection =
56 convectiveSchemeFromString(cfg.getString("numerics.convection", "muscl"));
57 options.limiter =
58 limiterFromString(cfg.getString("numerics.limiter", "vanLeer"));
59
60 const std::string initial = cfg.getString("heat.initial", "fourier");
61 if (initial == "fourier") {
62 if (options.source != 0)
63 throw ConfigError("saturnHeat: Fourier data requires heat.source = 0");
64 const Real amplitude = cfg.getDouble("heat.amplitude", 1.0);
65 const Real kx = cfg.getDouble("heat.kx", 6.283185307179586);
66 const Real ky = cfg.getDouble("heat.ky", 6.283185307179586);
67 const Real kz = cfg.getDouble("heat.kz", 6.283185307179586);
68 if (mesh.dimension < 2 && (ky != 0 || kz != 0))
69 throw ConfigError(
70 "saturnHeat: 1D Fourier data requires heat.ky = heat.kz = 0");
71 if (mesh.dimension < 3 && kz != 0)
72 throw ConfigError("saturnHeat: 2D Fourier data requires heat.kz = 0");
73 const Real k2 = kx * kx + ky * ky + kz * kz;
74 if (!(k2 > 0))
75 throw ConfigError("saturnHeat: Fourier wave vector must be nonzero");
76 const Real omega = options.velocity.x * kx + options.velocity.y * ky +
77 options.velocity.z * kz;
78 const Real decay = options.alpha * k2;
79 auto exact = [amplitude, kx, ky, kz, omega, decay](const Vec3& x, Real t) {
80 return amplitude * std::sin(kx * x.x + ky * x.y + kz * x.z - omega * t) *
81 std::exp(-decay * t);
82 };
83 options.initial = [exact](const Vec3& x) { return exact(x, 0); };
84 options.dirichlet = [exact](const Vec3& x, Real t, int) {
85 return exact(x, t);
86 };
87 options.exact = exact;
88 } else if (initial == "constant") {
89 const Real value = cfg.getDouble("heat.value", 1.0);
90 options.initial = [value](const Vec3&) { return value; };
91 if (!options.periodic)
92 options.dirichlet = [value](const Vec3&, Real, int) { return value; };
93 if (!options.periodic)
94 options.exact = [value](const Vec3&, Real) { return value; };
95 } else {
96 throw ConfigError("unsupported heat.initial '" + initial + "'");
97 }
98
99 return std::make_unique<SaturnHeat>(std::move(mesh), std::move(options));
100}
