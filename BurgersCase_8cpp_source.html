31Real travelingWave(Real x, Real t, Real speed, Real nu, Real x0) {
32 return Real(0.5) * speed *
33 (Real(1) -
34 std::tanh(speed * (x - x0 - Real(0.5) * speed * t) / (Real(4) * nu)));
35}

39ConfigSchema burgersSchema() {

60std::unique_ptr<Solver> makeSaturnBurger(const Config& cfg) {
61 Mesh mesh = makeMeshFromConfig(cfg);
62
63 BurgersOptions options;
64 options.nu = cfg.getDouble("burgers.nu", 0.02);
65 options.source = cfg.getDouble("burgers.source", 0.0);
66 options.periodic = cfg.getBool("burgers.periodic", false);
67 options.convection =
68 convectiveSchemeFromString(cfg.getString("numerics.convection", "muscl"));
69 options.limiter =
70 limiterFromString(cfg.getString("numerics.limiter", "vanLeer"));
71
72 const std::string initial = cfg.getString("burgers.initial", "travelingWave");
73 const Real speed = cfg.getDouble("burgers.velocity", 1.0);
74 const Real x0 = cfg.getDouble("burgers.x0", 0.0);
75
76 if (initial == "travelingWave") {
77 if (!(options.nu > 0))
78 throw ConfigError(
79 "burgers.initial: travelingWave requires burgers.nu > 0");
80 const Real nu = options.nu;
81 options.initial = [speed, nu, x0](const Vec3& x) {
82 return travelingWave(x.x, 0, speed, nu, x0);
83 };
84 options.dirichlet = [speed, nu, x0](const Vec3& x, Real t, int) {
85 return travelingWave(x.x, t, speed, nu, x0);
86 };
87 options.exact = [speed, nu, x0](const Vec3& x, Real t) {
88 return travelingWave(x.x, t, speed, nu, x0);
89 };
90 } else if (initial == "sine") {
91 if (!options.periodic)
92 throw ConfigError(
93 "burgers.initial: sine requires burgers.periodic: true");
94 options.initial = [](const Vec3& x) { return -std::sin(kPi * x.x); };
95 } else if (initial == "constant") {
96 const Real value = cfg.getDouble("burgers.value", 1.0);
97 options.initial = [value](const Vec3&) { return value; };
98 if (!options.periodic)
99 options.dirichlet = [value](const Vec3&, Real, int) { return value; };
100 } else {
101 throw ConfigError("unsupported burgers.initial '" + initial + "'");
102 }
103
104 return std::make_unique<SaturnBurger>(std::move(mesh), std::move(options));
105}
